   - Categorieen   : de categorieën in winkelvolgorde (bovenste rij = eerst)

   AI-categorisering (optioneel):
   - Haal een gratis Gemini API-sleutel op via https://aistudio.google.com/apikey

const SHEET_NAME  = 'Recepten';
const ING_SHEET   = 'Ingredienten';
const CAT_SHEET   = 'Categorieen';
const STATUS_SHEET = 'Status';   // gedeeld weekplan + boodschappen (JSON in cel A1)

function getSheet_() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sh = ss.getSheetByName(SHEET_NAME);
  if (!sh) {
    sh = ss.insertSheet(SHEET_NAME);
    sh.appendRow(['id', 'name', 'servings', 'photo', 'steps', 'ingredients_json', 'weight']);
    sh.setFrozenRows(1);
  }
  return sh;
}

function getIngSheet_() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sh = ss.getSheetByName(ING_SHEET);
  if (!sh) {
    sh = ss.insertSheet(ING_SHEET);
    sh.appendRow(['ingredient', 'categorie']);
    sh.setFrozenRows(1);
  }
  return sh;
}

function getCatSheet_() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sh = ss.getSheetByName(CAT_SHEET);
  if (!sh) {
    sh = ss.insertSheet(CAT_SHEET);
    sh.appendRow(['categorie']);
    sh.setFrozenRows(1);
  }
  return sh;
}

function getStatusSheet_() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sh = ss.getSheetByName(STATUS_SHEET);
  if (!sh) {
    sh = ss.insertSheet(STATUS_SHEET);
    sh.getRange(1, 1).setValue('');   // JSON komt in cel A1
  }
  return sh;
}

/* gedeelde status (weekplan, extra dingen, vinkjes) lezen/schrijven */
function readState_() {
  const sh = getStatusSheet_();
  const v = sh.getRange(1, 1).getValue();
  if (!v) return null;
  try { return JSON.parse(v); } catch (e) { return null; }
}
function writeState_(obj) {
  getStatusSheet_().getRange(1, 1).setValue(JSON.stringify(obj || {}));
}

/* voeg categorieën die in Ingredienten gebruikt worden maar nog niet in het
   Categorieen-tabblad staan achteraan toe (verwijdert niets — jij beheert de lijst). */
function syncCategoriesFromIngredients_() {
  const ingSh = getIngSheet_();
  const vals = ingSh.getDataRange().getValues();
  const used = [];
  for (let i = 1; i < vals.length; i++) {
    const c = String(vals[i][1] || '').trim();
    if (c && used.indexOf(c) < 0) used.push(c);
  }
  const catSh = getCatSheet_();
  const cvals = catSh.getDataRange().getValues();
  const have = {};
  for (let i = 1; i < cvals.length; i++) {
    const c = String(cvals[i][0] || '').trim();
    if (c) have[c.toLowerCase()] = true;
  }
  const addC = used.filter(c => !have[c.toLowerCase()]).map(c => [c]);
  if (addC.length) catSh.getRange(catSh.getLastRow() + 1, 1, addC.length, 1).setValues(addC);
}

/* ---------- AI-config (eigen categorieën + prompt) ---------- */
const AI_SHEET = 'AI_config';
const DEFAULT_PROMPT =
  'Je deelt boodschappen in supermarktcategorieën in.\n' +
  'Kies voor elk ingrediënt UITSLUITEND de best passende categorie uit deze lijst:\n' +
  '{categories}\n' +
  'Antwoord ALLEEN met een JSON-object van de vorm {"ingrediënt":"categorie", ...} ' +
  'met exact deze ingrediënten als sleutels:\n{ingredienten}';

function getAiSheet_() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sh = ss.getSheetByName(AI_SHEET);
  if (!sh) {
    sh = ss.insertSheet(AI_SHEET);
    sh.getRange(1, 1).setValue('prompt');
    sh.getRange(1, 2).setValue('');   // B1 = eigen prompt (leeg = standaard)
  }
  return sh;
}
function readPrompt_() {
  const v = getAiSheet_().getRange(1, 2).getValue();
  return v ? String(v) : '';
}
function writePrompt_(p) {
  getAiSheet_().getRange(1, 2).setValue(p || '');
}

/* roep Gemini Flash aan; retourneert de tekst (verwacht JSON) */
function callGemini_(prompt) {
  const key = PropertiesService.getScriptProperties().getProperty('GEMINI_API_KEY');
  if (!key) throw new Error('Geen GEMINI_API_KEY ingesteld (Projectinstellingen → Scripteigenschappen).');
  const model = 'gemini-2.5-flash';
  const url = 'https://generativelanguage.googleapis.com/v1beta/models/' + model + ':generateContent?key=' + encodeURIComponent(key);
  const payload = {
    contents: [{ parts: [{ text: prompt }] }],
    generationConfig: { responseMimeType: 'application/json', temperature: 0 }
  };
  const res = UrlFetchApp.fetch(url, {
    method: 'post', contentType: 'application/json',
    payload: JSON.stringify(payload), muteHttpExceptions: true
  });
  const code = res.getResponseCode();
  const txt = res.getContentText();
  if (code !== 200) throw new Error('Gemini ' + code + ': ' + txt.slice(0, 300));
  const j = JSON.parse(txt);
  const out = j.candidates && j.candidates[0] && j.candidates[0].content &&
              j.candidates[0].content.parts && j.candidates[0].content.parts[0].text;
  if (!out) throw new Error('Leeg AI-antwoord');
  return out;
}

function json_(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

/* lees ingrediënt→categorie map + categorie-volgorde */
function readCategories_() {
  const ingSh = getIngSheet_();
  const ingVals = ingSh.getDataRange().getValues();
  const map = {};
  const usedCats = [];
  for (let i = 1; i < ingVals.length; i++) {
    const name = String(ingVals[i][0] || '').trim();
    const cat  = String(ingVals[i][1] || '').trim();
    if (!name) continue;
    map[name.toLowerCase()] = cat;
    if (cat && usedCats.indexOf(cat) < 0) usedCats.push(cat);
  }
  const catSh = getCatSheet_();
  const catVals = catSh.getDataRange().getValues();
  const order = [];
  for (let i = 1; i < catVals.length; i++) {
    const c = String(catVals[i][0] || '').trim();
    if (c && order.indexOf(c) < 0) order.push(c);
  }
  // categorieën die wel gebruikt zijn maar nog niet geordend: achteraan toevoegen
  usedCats.forEach(c => { if (order.indexOf(c) < 0) order.push(c); });
  return { categories: map, categoryOrder: order };
}

/* ---------- READ ---------- */
function doGet(e) {
  try {
    // lichte poll: alleen de gedeelde status (gebruikt om te synchroniseren)
    if (e && e.parameter && e.parameter.action === 'state') {
      return json_({ ok: true, state: readState_() });
    }
    const sh = getSheet_();
    const values = sh.getDataRange().getValues();
    const recipes = [];
    for (let i = 1; i < values.length; i++) {
      const [id, name, servings, photo, steps, ingJson, weight] = values[i];
      if (!id) continue;
      let ingredients = [];
      try { ingredients = JSON.parse(ingJson || '[]'); } catch (err) {}
      recipes.push({
        id: String(id),
        name: String(name || ''),
        servings: Number(servings) || 1,
        photo: String(photo || ''),
        steps: String(steps || ''),
        ingredients: ingredients,
        weight: (weight === '' || weight == null) ? 5 : Number(weight)
      });
    }
    const cat = readCategories_();
    return json_({ ok: true, recipes: recipes, categories: cat.categories, categoryOrder: cat.categoryOrder, state: readState_(), aiPrompt: readPrompt_() });
  } catch (err) {
    return json_({ ok: false, error: String(err) });
  }
}

/* ---------- WRITE (save / delete / syncIngredients) ---------- */
function doPost(e) {
  try {
    const body = JSON.parse(e.postData.contents);

    if (body.action === 'save') {
      const sh = getSheet_();
      const r = body.recipe;
      const data = sh.getDataRange().getValues();
      let rowIndex = -1;
      for (let i = 1; i < data.length; i++) {
        if (String(data[i][0]) === String(r.id)) { rowIndex = i + 1; break; }
      }
      const row = [
        r.id,
        r.name,
        r.servings,
        r.photo || '',
        r.steps || '',
        JSON.stringify(r.ingredients || []),
        (r.weight === '' || r.weight == null) ? 5 : Number(r.weight)
      ];
      if (rowIndex === -1) sh.appendRow(row);
      else sh.getRange(rowIndex, 1, 1, row.length).setValues([row]);
      return json_({ ok: true });
    }

    if (body.action === 'saveState') {
      writeState_(body.state || {});
      return json_({ ok: true });
    }

    /* eigen categorieënlijst (= winkelvolgorde) + AI-prompt opslaan */
    if (body.action === 'saveAiConfig') {
      if (Array.isArray(body.categories)) {
        const seen = {};
        const list = [];
        body.categories.forEach(c => {
          const t = String(c || '').trim();
          const k = t.toLowerCase();
          if (t && !seen[k]) { seen[k] = true; list.push(t); }
        });
        const catSh = getCatSheet_();
        const last = catSh.getLastRow();
        if (last > 1) catSh.getRange(2, 1, last - 1, 1).clearContent();
        if (list.length) catSh.getRange(2, 1, list.length, 1).setValues(list.map(c => [c]));
      }
      if (typeof body.prompt === 'string') writePrompt_(body.prompt);
      return json_({ ok: true });
    }

    if (body.action === 'delete') {
      const sh = getSheet_();
      const data = sh.getDataRange().getValues();
      for (let i = 1; i < data.length; i++) {
        if (String(data[i][0]) === String(body.id)) {
          sh.deleteRow(i + 1);
          return json_({ ok: true });
        }
      }
      return json_({ ok: true }); // already gone
    }

    /* zet alle ingrediëntnamen in het Ingredienten-tabblad (zonder dubbele),
       en zet gebruikte categorieën in het Categorieen-tabblad */
    if (body.action === 'syncIngredients') {
      const names = body.names || [];
      const ingSh = getIngSheet_();
      let vals = ingSh.getDataRange().getValues();
      const have = {};
      for (let i = 1; i < vals.length; i++) {
        const n = String(vals[i][0] || '').trim().toLowerCase();
        if (n) have[n] = true;
      }
      const toAdd = [];
      names.forEach(n => {
        const t = String(n || '').trim();
        const k = t.toLowerCase();
        if (t && !have[k]) { have[k] = true; toAdd.push([t, '']); }
      });
      if (toAdd.length) {
        ingSh.getRange(ingSh.getLastRow() + 1, 1, toAdd.length, 2).setValues(toAdd);
      }
      syncCategoriesFromIngredients_();
      return json_({ ok: true, added: toAdd.length });
    }

    /* laat Gemini alle ingrediënten zonder categorie indelen */
    if (body.action === 'categorize') {
      const ingSh = getIngSheet_();
      const vals = ingSh.getDataRange().getValues();
      const todo = [];          // {row, name}
      const existingCats = [];
      for (let i = 1; i < vals.length; i++) {
        const name = String(vals[i][0] || '').trim();
        const cat  = String(vals[i][1] || '').trim();
        if (!name) continue;
        if (cat) { if (existingCats.indexOf(cat) < 0) existingCats.push(cat); }
        else todo.push({ row: i + 1, name: name });
      }
      if (!todo.length) return json_({ ok: true, filled: 0 });

      // toegestane categorieën = jouw lijst in het Categorieen-tabblad (anders standaardlijst)
      const order = readCategories_().categoryOrder;
      const defaults = ['Groente & fruit', 'Brood & banket', 'Zuivel & eieren', 'Kaas',
        'Vlees & vis', 'Diepvries', 'Pasta, rijst & granen', 'Conserven & houdbaar',
        'Sauzen, olie & azijn', 'Kruiden & specerijen', 'Ontbijt & broodbeleg',
        'Snoep, koek & snacks', 'Dranken', 'Non-food', 'Overig'];
      const catList = order.length ? order : defaults;

      const tmpl = readPrompt_() || DEFAULT_PROMPT;
      const prompt = tmpl
        .replace('{categories}', JSON.stringify(catList))
        .replace('{ingredienten}', JSON.stringify(todo.map(t => t.name)));

      const out = callGemini_(prompt);
      let mapping = {};
      try { mapping = JSON.parse(out); }
      catch (e) { throw new Error('Onverwacht AI-antwoord: ' + String(out).slice(0, 200)); }

      let filled = 0;
      todo.forEach(t => {
        const cat = mapping[t.name];
        if (cat && String(cat).trim()) {
          ingSh.getRange(t.row, 2).setValue(String(cat).trim());
          filled++;
        }
      });
      syncCategoriesFromIngredients_();
      return json_({ ok: true, filled: filled });
    }

    return json_({ ok: false, error: 'unknown action' });
  } catch (err) {
    return json_({ ok: false, error: String(err) });
  }
}
