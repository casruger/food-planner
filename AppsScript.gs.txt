   Tabbladen (worden automatisch aangemaakt):
   - Recepten      : de recepten zelf
   - Ingredienten  : elk ingrediënt + de categorie (winkel-stuk)
   - Categorieen   : de categorieën in winkelvolgorde (bovenste rij = eerst)

const SHEET_NAME = 'Recepten';
const ING_SHEET  = 'Ingredienten';
const CAT_SHEET  = 'Categorieen';

function getSheet_() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sh = ss.getSheetByName(SHEET_NAME);
  if (!sh) {
    sh = ss.insertSheet(SHEET_NAME);
    sh.appendRow(['id', 'name', 'servings', 'photo', 'steps', 'ingredients_json']);
    sh.setFrozenRows(1);
  }
  return sh;
}

function getIngSheet_() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sh = ss.getSheetByName(ING_SHEET);
  if (!sh) {
    sh = ss.insertSheet(ING_SHEET);
    sh.appendRow(['ingredient', 'categorie']);
    sh.setFrozenRows(1);
  }
  return sh;
}

function getCatSheet_() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sh = ss.getSheetByName(CAT_SHEET);
  if (!sh) {
    sh = ss.insertSheet(CAT_SHEET);
    sh.appendRow(['categorie']);
    sh.setFrozenRows(1);
  }
  return sh;
}

function json_(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

/* lees ingrediënt→categorie map + categorie-volgorde */
function readCategories_() {
  const ingSh = getIngSheet_();
  const ingVals = ingSh.getDataRange().getValues();
  const map = {};
  const usedCats = [];
  for (let i = 1; i < ingVals.length; i++) {
    const name = String(ingVals[i][0] || '').trim();
    const cat  = String(ingVals[i][1] || '').trim();
    if (!name) continue;
    map[name.toLowerCase()] = cat;
    if (cat && usedCats.indexOf(cat) < 0) usedCats.push(cat);
  }
  const catSh = getCatSheet_();
  const catVals = catSh.getDataRange().getValues();
  const order = [];
  for (let i = 1; i < catVals.length; i++) {
    const c = String(catVals[i][0] || '').trim();
    if (c && order.indexOf(c) < 0) order.push(c);
  }
  // categorieën die wel gebruikt zijn maar nog niet geordend: achteraan toevoegen
  usedCats.forEach(c => { if (order.indexOf(c) < 0) order.push(c); });
  return { categories: map, categoryOrder: order };
}

/* ---------- READ ---------- */
function doGet(e) {
  try {
    const sh = getSheet_();
    const values = sh.getDataRange().getValues();
    const recipes = [];
    for (let i = 1; i < values.length; i++) {
      const [id, name, servings, photo, steps, ingJson] = values[i];
      if (!id) continue;
      let ingredients = [];
      try { ingredients = JSON.parse(ingJson || '[]'); } catch (err) {}
      recipes.push({
        id: String(id),
        name: String(name || ''),
        servings: Number(servings) || 1,
        photo: String(photo || ''),
        steps: String(steps || ''),
        ingredients: ingredients
      });
    }
    const cat = readCategories_();
    return json_({ ok: true, recipes: recipes, categories: cat.categories, categoryOrder: cat.categoryOrder });
  } catch (err) {
    return json_({ ok: false, error: String(err) });
  }
}

/* ---------- WRITE (save / delete / syncIngredients) ---------- */
function doPost(e) {
  try {
    const body = JSON.parse(e.postData.contents);

    if (body.action === 'save') {
      const sh = getSheet_();
      const r = body.recipe;
      const data = sh.getDataRange().getValues();
      let rowIndex = -1;
      for (let i = 1; i < data.length; i++) {
        if (String(data[i][0]) === String(r.id)) { rowIndex = i + 1; break; }
      }
      const row = [
        r.id,
        r.name,
        r.servings,
        r.photo || '',
        r.steps || '',
        JSON.stringify(r.ingredients || [])
      ];
      if (rowIndex === -1) sh.appendRow(row);
      else sh.getRange(rowIndex, 1, 1, row.length).setValues([row]);
      return json_({ ok: true });
    }

    if (body.action === 'delete') {
      const sh = getSheet_();
      const data = sh.getDataRange().getValues();
      for (let i = 1; i < data.length; i++) {
        if (String(data[i][0]) === String(body.id)) {
          sh.deleteRow(i + 1);
          return json_({ ok: true });
        }
      }
      return json_({ ok: true }); // already gone
    }

    /* zet alle ingrediëntnamen in het Ingredienten-tabblad (zonder dubbele),
       en zet gebruikte categorieën in het Categorieen-tabblad */
    if (body.action === 'syncIngredients') {
      const names = body.names || [];
      const ingSh = getIngSheet_();
      let vals = ingSh.getDataRange().getValues();
      const have = {};
      for (let i = 1; i < vals.length; i++) {
        const n = String(vals[i][0] || '').trim().toLowerCase();
        if (n) have[n] = true;
      }
      const toAdd = [];
      names.forEach(n => {
        const t = String(n || '').trim();
        const k = t.toLowerCase();
        if (t && !have[k]) { have[k] = true; toAdd.push([t, '']); }
      });
      if (toAdd.length) {
        ingSh.getRange(ingSh.getLastRow() + 1, 1, toAdd.length, 2).setValues(toAdd);
      }

      // herbouw Categorieen-tabblad: behoud bestaande volgorde, verwijder
      // categorieën die nergens meer gebruikt worden, voeg nieuwe achteraan toe
      vals = ingSh.getDataRange().getValues();
      const usedCats = [];
      for (let i = 1; i < vals.length; i++) {
        const c = String(vals[i][1] || '').trim();
        if (c && usedCats.indexOf(c) < 0) usedCats.push(c);
      }
      const usedLower = {};
      usedCats.forEach(c => { usedLower[c.toLowerCase()] = true; });

      const catSh = getCatSheet_();
      const cvals = catSh.getDataRange().getValues();
      const newOrder = [];
      const seen = {};
      for (let i = 1; i < cvals.length; i++) {        // bestaande, nog gebruikte volgorde
        const c = String(cvals[i][0] || '').trim();
        const k = c.toLowerCase();
        if (c && usedLower[k] && !seen[k]) { seen[k] = true; newOrder.push(c); }
      }
      usedCats.forEach(c => {                          // nieuwe categorieën achteraan
        const k = c.toLowerCase();
        if (!seen[k]) { seen[k] = true; newOrder.push(c); }
      });

      const lastC = catSh.getLastRow();
      if (lastC > 1) catSh.getRange(2, 1, lastC - 1, 1).clearContent();
      if (newOrder.length) catSh.getRange(2, 1, newOrder.length, 1).setValues(newOrder.map(c => [c]));

      return json_({ ok: true, added: toAdd.length });
    }

    return json_({ ok: false, error: 'unknown action' });
  } catch (err) {
    return json_({ ok: false, error: String(err) });
  }
}
